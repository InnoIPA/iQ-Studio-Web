from iqs import Launcher

# one tag pulls, configures and runs
app = Launcher("iqs-vlm-demo")

app.source("/dev/video0")
app.model("qwen3-vl-4b", runtime="npu")

for frame, caption in app.stream():
    display(frame, caption)

# no model conversion. no chip code.
# first demo in ~10 minutes.
app.run()
